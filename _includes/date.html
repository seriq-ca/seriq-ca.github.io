{%- comment -%} Locale-aware date. Jekyll has no built-in localized date filter, so month names come from _data/i18n/.yml. Usage: {% include date.html date=page.event_date lang=page.lang %} {%- endcomment -%} {%- assign t = site.data.i18n[include.lang] -%} {%- assign midx = include.date | date: "%-m" | plus: 0 | minus: 1 -%} {%- assign month = t.months[midx] -%} {%- if include.lang == "fr" -%} {{ include.date | date: "%-d" }} {{ month }} {{ include.date | date: "%Y" }} {%- else -%} {{ month }} {{ include.date | date: "%-d" }}, {{ include.date | date: "%Y" }} {%- endif -%}
